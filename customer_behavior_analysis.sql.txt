/* 1. Customer Distribution by Country */
SELECT 
  Country,
  COUNT(*) AS total_customers
FROM `portfolio_sql.ecommerce_churn`
GROUP BY Country
ORDER BY total_customers DESC;


/* 2. Average Membership Years */
SELECT 
  ROUND(AVG(Membership_Years),2) AS avg_membership_years
FROM `portfolio_sql.ecommerce_churn`;


/* 3. Engagement Segmentation */
SELECT 
  CASE
    WHEN Login_Frequency >= 20 THEN 'High Engagement'
    WHEN Login_Frequency >= 10 THEN 'Medium Engagement'
    ELSE 'Low Engagement'
  END AS engagement_level,
  COUNT(*) AS users
FROM `portfolio_sql.ecommerce_churn`
GROUP BY engagement_level
ORDER BY users DESC;


/* 4. Top Engaged Customers */
SELECT 
  Country,
  Login_Frequency,
  Membership_Years
FROM `portfolio_sql.ecommerce_churn`
ORDER BY Login_Frequency DESC
LIMIT 10;


/* 5. Rank Customers by Engagement (Window Function) */
SELECT 
  Country,
  Login_Frequency,
  Membership_Years,
  ROW_NUMBER() OVER (ORDER BY Login_Frequency DESC) AS engagement_rank
FROM `portfolio_sql.ecommerce_churn`
LIMIT 20;


/* 6. Average Login Frequency by Country */
SELECT 
  Country,
  ROUND(AVG(Login_Frequency),2) AS avg_login_frequency
FROM `portfolio_sql.ecommerce_churn`
GROUP BY Country
ORDER BY avg_login_frequency DESC;


/* 7. Engagement Segmentation with Percentage (Advanced Window) */
WITH engagement AS (
  SELECT 
    CASE
      WHEN Login_Frequency >= 20 THEN 'High Engagement'
      WHEN Login_Frequency >= 10 THEN 'Medium Engagement'
      ELSE 'Low Engagement'
    END AS engagement_level
  FROM `portfolio_sql.ecommerce_churn`
)
SELECT 
  engagement_level,
  COUNT(*) AS users,
  ROUND(COUNT(*) * 100.0 / SUM(COUNT(*)) OVER (), 2) AS percentage
FROM engagement
GROUP BY engagement_level
ORDER BY users DESC;


/* 8. Long-Term Customers (Membership > 3 years) */
SELECT 
  Country,
  Membership_Years,
  Login_Frequency
FROM `portfolio_sql.ecommerce_churn`
WHERE Membership_Years > 3
ORDER BY Membership_Years DESC;
